cmake_minimum_required(VERSION 3.16)
project(tpathexp VERSION 1.2.0 LANGUAGES C)

include(CTest)
include(GNUInstallDirs)
option(BUILD_SHARED_LIBS "Build the loadable Tcl module instead of a static archive" ON)
find_path(TCL_INCLUDE_PATH NAMES tcl.h PATH_SUFFIXES tcl9.1 tcl9.0 tcl8.6 tcl)
if(NOT TCL_INCLUDE_PATH)
    message(FATAL_ERROR "Tcl development headers are required")
endif()
file(STRINGS "${TCL_INCLUDE_PATH}/tcl.h" TCL_VERSION_LINE REGEX "^#[ \t]*define TCL_VERSION[ \t]+\"")
string(REGEX MATCH "[0-9]+\\.[0-9]+" TPATHEXP_TCL_VERSION "${TCL_VERSION_LINE}")
if(TPATHEXP_TCL_VERSION VERSION_LESS 8.6 OR NOT TPATHEXP_TCL_VERSION VERSION_LESS 10)
    message(FATAL_ERROR "Tcl 8.6 or 9.x is required")
endif()
if(TPATHEXP_TCL_VERSION VERSION_LESS 9)
    set(TPATHEXP_TCL_REQUIREMENT "8.6-9")
else()
    set(TPATHEXP_TCL_REQUIREMENT "9.0-10")
endif()
find_library(TCL_LIBRARY NAMES tcl${TPATHEXP_TCL_VERSION}
    HINTS "${TCL_INCLUDE_PATH}/../lib")
find_program(TCL_TCLSH NAMES tclsh${TPATHEXP_TCL_VERSION}
    HINTS "${TCL_INCLUDE_PATH}/../bin")
if(NOT TCL_LIBRARY)
    message(FATAL_ERROR "Library for Tcl ${TPATHEXP_TCL_VERSION} is required")
endif()
if(BUILD_SHARED_LIBS)
    get_filename_component(TCL_LIBRARY_DIR "${TCL_LIBRARY}" DIRECTORY)
    find_library(TCL_STUB_LIBRARY NAMES tclstub${TPATHEXP_TCL_VERSION} tclstub
        HINTS "${TCL_LIBRARY_DIR}" "${TCL_INCLUDE_PATH}/../lib" "${TCL_INCLUDE_PATH}/../../lib")
    if(NOT TCL_STUB_LIBRARY)
        message(FATAL_ERROR "The Tcl stub library is required for shared builds")
    endif()
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(tpathexp src/tpathexp.c src/router_tcl.c src/router.c src/path_regexp.c)
target_compile_definitions(tpathexp PRIVATE TPATHEXP_VERSION="${PROJECT_VERSION}")
if(BUILD_SHARED_LIBS)
    target_compile_definitions(tpathexp PRIVATE USE_TCL_STUBS)
    target_link_libraries(tpathexp PRIVATE ${TCL_STUB_LIBRARY})
    set(TPATHEXP_LOAD "[list load [file join $dir ${CMAKE_SHARED_LIBRARY_PREFIX}tpathexp${CMAKE_SHARED_LIBRARY_SUFFIX}] Tpathexp]")
else()
    set(TPATHEXP_LOAD "[list load {} Tpathexp]")
endif()
target_include_directories(tpathexp PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/tpathexp>
    ${TCL_INCLUDE_PATH})
set_target_properties(tpathexp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(tpathexp PRIVATE -Wall -Wextra -Wpedantic)
endif()

configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
set(TPATHEXP_PACKAGE_DIR "${CMAKE_INSTALL_LIBDIR}/tpathexp${PROJECT_VERSION}")
install(TARGETS tpathexp
    LIBRARY DESTINATION "${TPATHEXP_PACKAGE_DIR}"
    ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl" DESTINATION "${TPATHEXP_PACKAGE_DIR}")
install(FILES src/tpathexp.h src/path_regexp.h src/router.h DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/tpathexp")

if(BUILD_TESTING)
    add_executable(test_router tests/router.c)
    target_link_libraries(test_router PRIVATE tpathexp)
    add_test(NAME native_router COMMAND test_router)
    add_executable(example_router examples/router.c)
    target_link_libraries(example_router PRIVATE tpathexp)
    add_test(NAME native_example COMMAND example_router)
    if(BUILD_SHARED_LIBS)
        if(NOT TCL_TCLSH)
            message(FATAL_ERROR "Tcl ${TPATHEXP_TCL_VERSION} shell is required to run tests")
        endif()
        set(TPATHEXP_TEST_SHELL "${TCL_TCLSH}")
    else()
        add_executable(test_embed tests/embed.c)
        target_link_libraries(test_embed PRIVATE tpathexp ${TCL_LIBRARY})
        set(TPATHEXP_TEST_SHELL "$<TARGET_FILE:test_embed>")
        add_test(NAME static_embed COMMAND test_embed "${CMAKE_CURRENT_SOURCE_DIR}/tests/embed.tcl")
        set_tests_properties(static_embed PROPERTIES ENVIRONMENT "TCLLIBPATH=${CMAKE_CURRENT_BINARY_DIR}")
    endif()
    add_test(NAME tcl COMMAND ${TPATHEXP_TEST_SHELL} "${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl")
    set_tests_properties(tcl PROPERTIES ENVIRONMENT "TCLLIBPATH=${CMAKE_CURRENT_BINARY_DIR}")
    configure_file(tests/install.cmake.in tests/install.cmake @ONLY)
    add_test(NAME install_layout COMMAND ${CMAKE_COMMAND} -DCONFIG=$<CONFIG>
        -P "${CMAKE_CURRENT_BINARY_DIR}/tests/install.cmake")
endif()
