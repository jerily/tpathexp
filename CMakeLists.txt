cmake_minimum_required(VERSION 3.16)
project(tpathexp VERSION 1.0.0 LANGUAGES C)

include(CTest)
include(GNUInstallDirs)
find_path(TCL_INCLUDE_PATH NAMES tcl.h PATH_SUFFIXES tcl9.1 tcl9.0 tcl8.6 tcl)
if(NOT TCL_INCLUDE_PATH)
    message(FATAL_ERROR "Tcl development headers are required")
endif()
file(STRINGS "${TCL_INCLUDE_PATH}/tcl.h" TCL_VERSION_LINE REGEX "^#[ \t]*define TCL_VERSION[ \t]+\"")
string(REGEX MATCH "[0-9]+\\.[0-9]+" TPATHEXP_TCL_VERSION "${TCL_VERSION_LINE}")
if(TPATHEXP_TCL_VERSION VERSION_LESS 8.6 OR NOT TPATHEXP_TCL_VERSION VERSION_LESS 10)
    message(FATAL_ERROR "Tcl 8.6 or 9.x is required")
endif()
if(TPATHEXP_TCL_VERSION VERSION_LESS 9)
    set(TPATHEXP_TCL_REQUIREMENT "8.6-9")
else()
    set(TPATHEXP_TCL_REQUIREMENT "9.0-10")
endif()
find_library(TCL_LIBRARY NAMES tcl${TPATHEXP_TCL_VERSION}
    HINTS "${TCL_INCLUDE_PATH}/../lib")
find_program(TCL_TCLSH NAMES tclsh${TPATHEXP_TCL_VERSION}
    HINTS "${TCL_INCLUDE_PATH}/../bin")
if(NOT TCL_LIBRARY)
    message(FATAL_ERROR "Library for Tcl ${TPATHEXP_TCL_VERSION} is required")
endif()
get_filename_component(TCL_LIBRARY_DIR "${TCL_LIBRARY}" DIRECTORY)
find_library(TCL_STUB_LIBRARY NAMES tclstub${TPATHEXP_TCL_VERSION} tclstub
    HINTS "${TCL_LIBRARY_DIR}" "${TCL_INCLUDE_PATH}/../lib" "${TCL_INCLUDE_PATH}/../../lib")
if(NOT TCL_STUB_LIBRARY)
    message(FATAL_ERROR "Tcl development files, including the stub library, are required")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(TPATHEXP_SOURCES src/tpathexp.c src/path_regexp.c)
add_library(tpathexp SHARED ${TPATHEXP_SOURCES})
target_compile_definitions(tpathexp PRIVATE USE_TCL_STUBS)
target_link_libraries(tpathexp PRIVATE ${TCL_STUB_LIBRARY})

add_library(tpathexp_static STATIC ${TPATHEXP_SOURCES})
set_target_properties(tpathexp_static PROPERTIES OUTPUT_NAME tpathexp)
target_link_libraries(tpathexp_static PUBLIC ${TCL_LIBRARY})
foreach(target tpathexp tpathexp_static)
    target_include_directories(${target} PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/tpathexp>
        ${TCL_INCLUDE_PATH})
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
configure_file(tcl/tpathexp.tcl tpathexp.tcl COPYONLY)
set(TPATHEXP_PACKAGE_DIR "${CMAKE_INSTALL_LIBDIR}/tpathexp${PROJECT_VERSION}")
install(TARGETS tpathexp LIBRARY DESTINATION "${TPATHEXP_PACKAGE_DIR}")
install(TARGETS tpathexp_static ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl" tcl/tpathexp.tcl
    DESTINATION "${TPATHEXP_PACKAGE_DIR}")
install(FILES src/tpathexp.h src/path_regexp.h DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/tpathexp")

if(BUILD_TESTING)
    if(NOT TCL_TCLSH)
        message(FATAL_ERROR "Tcl ${TPATHEXP_TCL_VERSION} shell is required to run tests")
    endif()
    add_test(NAME tcl COMMAND "${TCL_TCLSH}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl")
    set_tests_properties(tcl PROPERTIES ENVIRONMENT "TCLLIBPATH=${CMAKE_CURRENT_BINARY_DIR}")
    add_executable(test_embed tests/embed.c)
    target_link_libraries(test_embed PRIVATE tpathexp_static)
    add_test(NAME static_embed COMMAND test_embed "${CMAKE_CURRENT_BINARY_DIR}/tpathexp.tcl")
endif()
